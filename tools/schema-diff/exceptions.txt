# que matcheen se filtran ANTES del diff, en ambos lados. Cada patrón debe

# --- Tablas propias del framework Laravel (no existen en Supabase) ---
^[a-z]+\|(migrations|password_reset_tokens|sessions|cache|cache_locks|jobs|job_batches|failed_jobs)(\||$)
^[a-z]+\|(users|passkeys|api_key_requests)(\||$)

# --- Tablas del ai-sdk de Laravel (laravel/ai) ---
^[a-z]+\|(agent_conversations|agent_conversation_messages)(\||$)

# --- Familia auth/tenancy: rediseñada Laravel-style en #12-#21 ---
# profiles -> users (bigint); account_members -> account_user; accounts,
# account_invitations y api_keys se reconstruyeron con tipos Laravel
# (varchar, timestamp sin tz, FKs bigint a users). Ver docs/schema-deviations.md.
^[a-z]+\|(profiles|account_members|accounts|account_user|account_invitations|api_keys)(\||$)

# --- uuid-ossp: Laravel usa gen_random_uuid()/HasUuids, no la extensión ---
^extension\|uuid-ossp$

# --- FKs a usuarios en dominio core (#39): uuid (auth.users/profiles) ->
# bigint (users). Patrones anclados por tabla+columna+nullabilidad para no
# tapar otras diferencias. Ver docs/schema-deviations.md. ---
^column\|(contacts|tags|custom_fields|contact_notes|conversations|whatsapp_config|message_templates|pipelines|deals|quick_replies|notifications|member_presence)\|user_id\|(uuid|bigint)\|not null\|-\|-$
^column\|conversations\|assigned_agent_id\|(uuid|bigint)\|null\|-\|-$
^column\|deals\|assigned_to\|(uuid|bigint)\|null\|-\|-$
^column\|messages\|sender_id\|(uuid|bigint)\|null\|-\|-$
^column\|message_reactions\|actor_id\|(uuid|bigint)\|null\|-\|-$
^column\|notifications\|actor_user_id\|(uuid|bigint)\|null\|-\|-$
^constraint\|(contacts|tags|custom_fields|contact_notes|conversations|whatsapp_config|message_templates|pipelines|deals|quick_replies|notifications|member_presence)\|FOREIGN KEY \((user_id|actor_user_id)\) REFERENCES (auth\.)?users\(id\) ON DELETE (CASCADE|SET NULL)$
^constraint\|deals\|FOREIGN KEY \(assigned_to\) REFERENCES (profiles|users)\(id\) ON DELETE SET NULL$

# --- FKs a usuarios en motores + IA (#40): misma familia uuid -> bigint ---
^column\|(broadcasts|automations|automation_logs|automation_pending_executions|flows|flow_runs)\|user_id\|(uuid|bigint)\|not null\|-\|-$
^column\|(webhook_endpoints|ai_configs|ai_knowledge_documents)\|created_by\|(uuid|bigint)\|null\|-\|-$
^column\|ai_configs\|handoff_agent_id\|(uuid|bigint)\|null\|-\|-$
^constraint\|(broadcasts|automations|automation_logs|automation_pending_executions|flows|flow_runs)\|FOREIGN KEY \(user_id\) REFERENCES (auth\.)?users\(id\) ON DELETE CASCADE$
^constraint\|(webhook_endpoints|ai_configs|ai_knowledge_documents)\|FOREIGN KEY \(created_by\) REFERENCES (auth\.)?users\(id\) ON DELETE SET NULL$
^constraint\|ai_configs\|FOREIGN KEY \(handoff_agent_id\) REFERENCES (auth\.)?users\(id\) ON DELETE SET NULL$

# --- notifications/quick_replies.id: Supabase usaba uuid_generate_v4()
# (uuid-ossp, 027/035); Laravel usa gen_random_uuid() nativo (#39) ---
^column\|(notifications|quick_replies)\|id\|uuid\|not null\|(uuid_generate_v4|gen_random_uuid)\(\)\|-$
